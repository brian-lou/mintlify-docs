使用 Steps 来展示一系列按顺序进行的操作或事件。你可以根据需要添加任意数量的步骤。

<Steps>
  <Step title="第一步">
    这些说明或内容仅与第一步相关。
  </Step>

  <Step title="第二步">
    这些说明或内容仅与第二步相关。
  </Step>

  <Step title="第三步">
    这些说明或内容仅与第三步相关。
  </Step>
</Steps>

```mdx Steps example
<Steps>
  <Step title="第一步">
    这些是仅与第一步相关的说明或内容。
  </Step>
  <Step title="第二步">
    这些是仅与第二步相关的说明或内容。
  </Step>
  <Step title="第三步">
    这些是仅与第三步相关的说明或内容。
  </Step>
</Steps>
```


<div id="steps-properties">
  ## 步骤属性
</div>

<ResponseField name="children" type="ReactElement<StepProps>[]" required>
  一个 `Step` 组件列表。
</ResponseField>

<ResponseField name="titleSize" type="string" default="p">
  步骤标题的字号。可选值为 `p`、`h2` 和 `h3`。
</ResponseField>

<div id="individual-step-properties">
  ## 每个步骤的属性
</div>

<ResponseField name="children" type="string | ReactNode" required>
  步骤的内容，可以是纯文本或组件。
</ResponseField>

<IconsOptional />

<ResponseField name="title" type="string">
  标题是该步骤的主要文本，会显示在指示器旁边。
</ResponseField>

<ResponseField name="stepNumber" type="number">
  步骤的序号。
</ResponseField>

<ResponseField name="titleSize" type="string" default="p">
  步骤标题的大小。可选值为 `p`、`h2` 和 `h3`。
</ResponseField>
